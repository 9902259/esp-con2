cmake_minimum_required(VERSION 3.20)
# Use the pinned local espp checkout, including its NimBLE submodule.
set(ENV{IDF_COMPONENT_MANAGER} "0")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../.deps/espp/components")
if("$ENV{ESP_CON2_PROFILE}" MATCHES "^(joycon2-(left|right|dual|bootstrap)|unified)$")
  # Explicit dependency directories avoid registering two drivers with the same name.
  set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../.deps/$ENV{ESP_CON2_PROFILE}/switch2_pro")
  if("$ENV{ESP_CON2_PROFILE}" STREQUAL "unified")
    set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../.deps/joycon2-dual/switch2_pro"
      "${CMAKE_CURRENT_LIST_DIR}/../.deps/unified/unified_pro")
    add_compile_definitions(ESP_CON2_UNIFIED=1 ESP_CON2_JOYCON2_DUAL=1)
  endif()
  foreach(component base_component logger format task timer nvs cli ble_gatt_server esp-nimble-cpp)
    list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../.deps/espp/components/${component}")
  endforeach()
  if("$ENV{ESP_CON2_PROFILE}" MATCHES "^(joycon2-dual|unified)$")
    add_compile_definitions(ESP_CON2_JOYCON2_DUAL=1)
  elseif("$ENV{ESP_CON2_PROFILE}" STREQUAL "joycon2-bootstrap")
    add_compile_definitions(ESP_CON2_JOYCON2_LEFT=1 ESP_CON2_DUAL_BOOTSTRAP=1)
  elseif("$ENV{ESP_CON2_PROFILE}" STREQUAL "joycon2-left")
    add_compile_definitions(ESP_CON2_JOYCON2_LEFT=1)
  else()
    add_compile_definitions(ESP_CON2_JOYCON2_RIGHT=1)
  endif()
endif()
set(COMPONENTS main esptool_py)
project(esp_con2)
if("$ENV{ESP_CON2_PROFILE}" MATCHES "^(joycon2-dual|unified)$")
  # Allow two controller identities to accept the same Switch central in this experimental profile.
  target_link_options(esp_con2.elf PRIVATE "-Wl,--wrap=r_llm_is_dev_connected")
endif()
set(CMAKE_CXX_STANDARD 23)
